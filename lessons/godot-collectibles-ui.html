03 / SIGNAL

One event, many listeners

Collectibles report. Systems decide.

collectible.gd
extends Area3D
signal collected(item_id: String, value: int)
@export var item_id := "energy_cell"
@export var value := 1

func _ready() -> void:
    body_entered.connect(_on_body_entered)

func _on_body_entered(body: Node3D) -> void:
    if body.is_in_group("player"):
        collected.emit(item_id, value)
        queue_free()

04 / DATA

Inventory before interface

Store facts once, redraw any UI.

inventory.gd · Autoload
extends Node
signal changed(items: Dictionary)
var items: Dictionary = {}

func add_item(item_id: String, amount: int = 1) -> void:
    items[item_id] = items.get(item_id, 0) + amount
    changed.emit(items)

func spend(item_id: String, amount: int) -> bool:
    if items.get(item_id, 0) < amount:
        return false
    items[item_id] -= amount
    changed.emit(items)
    return true
